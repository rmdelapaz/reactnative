labs/module-22/l8_4-performance-and-debugging/solution/src/SiteSearch.tsx
// A beach search for crew leaders: type part of a beach's name, set a low-tide reminder for one
import { useState } from 'react';
import { Pressable, StyleSheet, Text, TextInput, View } from 'react-native';

import sites from './data/sites.json';
import { remindMe } from './reminders';

export type Site = { id: string; name: string; town: string };

// The beaches whose name contains what the crew leader typed, whatever the case of the letters
export function matchingSites(all: Site[], query: string): Site[] {
  const wanted = query.trim().toLowerCase();
  return all.filter((site) => site.name.toLowerCase().includes(wanted));
}

export function SiteSearch() {
  const [query, setQuery] = useState('');
  const [reminded, setReminded] = useState<string | null>(null);
  const found = matchingSites(sites, query);

  return (
    <View style={styles.screen}>
      <TextInput
        accessibilityLabel="Search beaches"
        placeholder="Search beaches"
        value={query}
        onChangeText={setQuery}
        style={styles.field}
      />
      <Text>{found.length === 1 ? '1 beach' : `${found.length} beaches`}</Text>
      {found.map((site) => (
        <View key={site.id} style={styles.row}>
          <View style={styles.words}>
            <Text style={styles.name}>{site.name}</Text>
            <Text>{site.town}</Text>
          </View>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel={`Remind me about low tide at ${site.name}`}
            onPress={async () => {
              await remindMe(site.name);
              setReminded(site.name);
            }}
            style={styles.button}
          >
            <Text style={styles.buttonText}>Remind me</Text>
          </Pressable>
        </View>
      ))}
      {reminded !== null && <Text>Reminder set for {reminded}</Text>}
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, gap: 12, padding: 24, backgroundColor: '#fbf6ec' },
  field: { borderWidth: 1, borderColor: '#0b3954', borderRadius: 8, padding: 12, fontSize: 16 },
  row: { flexDirection: 'row', alignItems: 'center', gap: 12 },
  words: { flex: 1 },
  name: { fontSize: 16, fontWeight: '600', color: '#0b3954' },
  button: { minHeight: 48, justifyContent: 'center', paddingHorizontal: 12, borderRadius: 8, backgroundColor: '#1f6f8b' },
  buttonText: { color: '#ffffff', fontWeight: '600' },
});
labs/module-22/l8_4-performance-and-debugging/solution/src/useOnForeground.ts
import { useEffect } from 'react';
import { AppState } from 'react-native';

// Runs refresh each time the app comes back to the foreground (a crew leader returns from a call, the camera...)
export function useOnForeground(refresh: () => void) {
  useEffect(() => {
    const subscription = AppState.addEventListener('change', (state) => {
      if (state === 'active') refresh();
    });
    // The screen closes: stop listening, or every visit adds one more listener that never goes away
    return () => subscription.remove();
  }, [refresh]);
}

// Shoreline Lookout's app config: a crew leader's beach-conditions app (Lab 9.1), three variants of one project
import type { ConfigContext, ExpoConfig } from 'expo/config';

const variants = {
  development: { name: 'Lookout Dev', appId: 'com.example.shorelinelookout.dev' },
  preview: { name: 'Lookout Preview', appId: 'com.example.shorelinelookout.preview' },
  production: { name: 'Shoreline Lookout', appId: 'com.example.shorelinelookout' },
};
type Variant = keyof typeof variants;

function isVariant(value: string): value is Variant {
  return Object.hasOwn(variants, value);
}

// APP_VARIANT picks the variant; nothing set means development, so a forgotten setting never makes the store app
const variant = process.env.APP_VARIANT ?? 'development';
if (!isVariant(variant)) {
  throw new Error(`APP_VARIANT must be development, preview or production, not "${variant}".`);
}

// A store build must not talk plain http (release builds block it, and anyone on the Wi-Fi could read it)
const apiUrl = process.env.EXPO_PUBLIC_API_URL ?? '';
if (variant === 'production' && !apiUrl.startsWith('https://')) {
  throw new Error(`A production build needs an https:// EXPO_PUBLIC_API_URL, not "${apiUrl}".`);
}

export default ({ config }: ConfigContext): ExpoConfig => ({
  ...config,
  name: variants[variant].name,
  slug: 'shoreline-lookout',
  version: '1.0.0',
  // With the development build and the store app on one phone, a link should open the right one
  scheme: variant === 'development' ? 'shorelinelookout-dev' : 'shorelinelookout',
  ios: {
    bundleIdentifier: variants[variant].appId,
