import { render, screen, userEvent } from '@testing-library/react-native';
import { useState } from 'react';
import { FlatList, Text, TextInput } from 'react-native';

const beaches = ['Driftwood Cove', 'Otter Creek Mouth', 'Long Strand', 'Jetty Rocks'];
jest.setTimeout(30_000); // a first render on a busy laptop can take seconds

// Don't: a new header component on every render (Lesson 3.1, Common Mistakes)
function InlineHeader() {
  const [query, setQuery] = useState('');
  return (
    <FlatList
      data={beaches.filter((beach) => beach.toLowerCase().includes(query.toLowerCase()))}
      keyExtractor={(beach) => beach}
      renderItem={({ item }) => <Text>{item}</Text>}
      ListHeaderComponent={() => (
        <TextInput accessibilityLabel="Search beaches" value={query} onChangeText={setQuery} />
      )}
    />
  );
}

// Don't: the number of columns changes, but the FlatList is the same one
function NoKey({ columns }: { columns: number }) {
  return (
    <FlatList
      data={beaches}
      keyExtractor={(beach) => beach}
      numColumns={columns}
      renderItem={({ item }) => <Text>{item}</Text>}
    />
  );
}

test('an inline header: type "lo"', async () => {
  const user = userEvent.setup();
  await render(<InlineHeader />);
  await user.type(screen.getByLabelText('Search beaches'), 'lo');
  expect(screen.getByLabelText('Search beaches')).toHaveDisplayValue('lo');
});

test('numColumns from 1 to 2, no key', async () => {
  await render(<NoKey columns={1} />);
  await screen.rerender(<NoKey columns={2} />);
});

The solution's hook:

  // The next page, added after the cleanups the list already has
  function loadNextPage() {
    if (load.status !== 'loaded' || !hasMore) {
      return;
    }
    setMore({ status: 'loading' });
    getPastPage(api, load.page.page + 1)
      .then((next) => {
        setLoad((current) => {
          if (current.status !== 'loaded') {
            return current;
          }

          // A cleanup can move to the next page while you scroll: never show it twice
          const known = new Set(current.cleanups.map((cleanup) => cleanup.id));
          const added = next.items.filter((cleanup) => !known.has(cleanup.id));
          return { ...current, cleanups: [...current.cleanups, ...added], page: next.page };
        });
        setMore({ status: 'idle' });
      })
      .catch((error: unknown) => setMore({ status: 'failed', message: message(error) }));
  }

  // The list is near its end. Only ask when nothing is on its way and the last page did not fail:
  // after a failure, Try again (loadNextPage) is the only way to ask again
  function onEndReached() {
    if (more.status === 'idle') {
      loadNextPage();
    }
  }

  return { load, more, hasMore, loadNextPage, onEndReached };
}

The Lesson Pages' Own Quick Quizzes

Lesson 3.1 · FlatList and SectionList
  1. D The keyboard closes after every letter: The arrow function is a new component on every render, so React replaces the header, and its text field, after each letter. Focus is lost and the keyboard closes. Pass an element (const header = ...) instead.
  2. B An error: changing numColumns on the fly is not supported: FlatList cannot change numColumns in place: it stops with Changing numColumns on the fly is not supported. A key that changes with the number of columns, key={columns}, makes React build a new list instead.
  3. C (card) => card.event.id: The API's id is unique and never changes, so each drawn card stays with its cleanup while the list is filtered, refreshed or regrouped. Positions and titles can both change.
  4. A In the list's ListEmptyComponent, under the banner and the search field: The failed load leaves the list with no cards, so FlatList shows ListEmptyComponent, which is the ErrorView in the failed state. The header (banner, search, switches) stays above it.
  5. B SectionList has no numColumns: SectionList has no numColumns, so the grouped view is one card per line at any width. Getting columns there means putting rows of cards in each section yourself.
Lesson 3.2 · Big Lists, Fast
  1. B They are kept and reused to draw the cards coming into view: FlashList keeps them in a pool of spare rows of the same type and draws them again with the next items' data. That is why the lab counted a few dozen rows created for 1,000 items.
  2. D useRecyclingState(false, [event.id]): useRecyclingState resets to its initial value when a dependency changes. With the cleanup's id as the dependency, a recycled card starts closed for its new cleanup.
  3. B The footer shows the error and Try again; nothing is asked until the volunteer presses it: After a failure the next page is failed, not idle, so onEndReached does nothing. The footer shows describeError's message and Try again, which is the only thing that asks again.
  4. C The short list is near its end, so it loads page 2, where the cleanup is found: With no cards, the list is shorter than the screen, so it is near its end and calls onEndReached. Page 2 arrives, the filter runs again, and Night Heron Walk and Clean appears: 1 of 8 cleanups.
  5. C Half the gap of padding inside each cell, on both sides: FlashList positions every cell itself, so each cell carries half the gap on each side. The list's own side padding is halved, and the header, empty list and footer get the other half, so everything lines up.

Common Wrong Answers and What They Reveal

  • Q1 D. Thinks virtualization draws only what is visible. The lesson's 358: ten screens each way.
  • Q3 A. Fixes the type and keeps a non-unique key. Ask: can two cleanups have the same capacity?
  • Q5 B or C. Debugs the code when the platform is the difference. Show the web-only Refresh button.
  • Q7 B. Reads frame rates on a desktop as proof; full speed hides the gap. The 6x line and the rows created.
  • Q9 A. Expects tsc to know FlashList's documentation. It type-checks: that is why it is a trap.
  • Q12 A or B. Trusts page numbers as stable. Pages count from the start of the list, which can move.
