AppText from Lesson 8.1: live text is announced on iPhone too, and titles grow with the phone's text size only up to 1.75 times. Lesson 8.1, Lesson 8.1.

export function AppText({ variant = 'body', style, ...rest }: AppTextProps) {
  const styles = useStyles();
  const live = rest.accessibilityLiveRegion !== undefined && rest.accessibilityLiveRegion !== 'none';
  const words = live ? wordsOf(rest.children) : '';
  useEffect(() => {
    // iPhone has no live regions (accessibilityLiveRegion is for Android and the web): VoiceOver
    // says the new words because we ask it to, queued behind whatever it is saying already
    if (Platform.OS === 'ios' && words !== '') {
      AccessibilityInfo.announceForAccessibilityWithOptions(words, { queue: true });
    }
  }, [words]);
  return (
    <Text
      // Titles are big already: they grow with the phone's text size, but at most 1.75 times
      maxFontSizeMultiplier={variant === 'title' ? 1.75 : undefined}
      style={[styles.base, styles[variant], style]}
      {...rest}
    />
  );
}

Platform.select: one value per platform, with default for the rest (the web preview here). Lesson 2.5, Lesson 6.3.

export function mapLink({ latitude, longitude }: Coordinates, label: string): string {
  const at = `${latitude},${longitude}`;
  const name = encodeURIComponent(label);
  return Platform.select({
    ios: `https://maps.apple.com/?ll=${at}&q=${name}`,
    android: `geo:${at}?q=${at}(${name})`,
    default: `https://www.openstreetmap.org/?mlat=${latitude}&mlon=${longitude}#map=16/${latitude}/${longitude}`,
  });
}

Light and dark palettes, and styles made once per theme. Lesson 2.5.

// The theme for the phone's current setting: light, or dark when the phone is in dark mode
export function useTheme(): Theme {
  const scheme = useColorScheme();
  return themes[scheme === 'dark' ? 'dark' : 'light'];
}

// Styles that use theme colors: made once for each theme, then picked by the current one.
// Call it outside a component; it returns a hook to call inside one.
export function createThemedStyles<T>(make: (theme: Theme) => T) {
  const sheets = { light: make(themes.light), dark: make(themes.dark) };
  return function useStyles(): T {
    return sheets[useTheme().scheme];
  };
}
